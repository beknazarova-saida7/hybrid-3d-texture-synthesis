"""
Stage 3: CNN neural refinement.
A lightweight U-Net (~1.2M parameters) refines the warped procedural
texture using a combined style + content + total-variation loss computed
on VGG19 features (Eq. 8).

Requires: torch, torchvision.
"""
import torch
import torch.nn as nn
import torch.nn.functional as F
import torchvision.models as models


# ---------------------------------------------------------------- U-Net --
class ConvBlock(nn.Module):
    def __init__(self, in_ch, out_ch):
        super().__init__()
        self.block = nn.Sequential(
            nn.Conv2d(in_ch, out_ch, 3, padding=1), nn.InstanceNorm2d(out_ch), nn.ReLU(inplace=True),
            nn.Conv2d(out_ch, out_ch, 3, padding=1), nn.InstanceNorm2d(out_ch), nn.ReLU(inplace=True),
        )

    def forward(self, x):
        return self.block(x)


class LightUNet(nn.Module):
    """Lightweight U-Net with skip connections, ~1.2M parameters at base
    channel width 24 (matches the paper's Section III-C configuration)."""

    def __init__(self, in_ch=1, base=24):
        super().__init__()
        self.enc1 = ConvBlock(in_ch, base)
        self.enc2 = ConvBlock(base, base * 2)
        self.enc3 = ConvBlock(base * 2, base * 4)
        self.pool = nn.MaxPool2d(2)

        self.bottleneck = ConvBlock(base * 4, base * 4)

        self.up2 = nn.ConvTranspose2d(base * 4, base * 2, 2, stride=2)
        self.dec2 = ConvBlock(base * 4, base * 2)
        self.up1 = nn.ConvTranspose2d(base * 2, base, 2, stride=2)
        self.dec1 = ConvBlock(base * 2, base)

        self.out_conv = nn.Conv2d(base, in_ch, 1)

    def forward(self, x):
        e1 = self.enc1(x)
        e2 = self.enc2(self.pool(e1))
        e3 = self.enc3(self.pool(e2))
        b = self.bottleneck(e3)

        d2 = self.up2(b)
        d2 = self.dec2(torch.cat([d2, e2], dim=1))
        d1 = self.up1(d2)
        d1 = self.dec1(torch.cat([d1, e1], dim=1))
        return torch.sigmoid(self.out_conv(d1))


def count_parameters(model):
    return sum(p.numel() for p in model.parameters() if p.requires_grad)


# -------------------------------------------------------- VGG19 losses --
class VGGFeatures(nn.Module):
    LAYER_MAP = {"3": "relu1_1", "8": "relu2_1", "13": "relu3_1", "15": "relu3_3"}

    def __init__(self):
        super().__init__()
        vgg = models.vgg19(weights=models.VGG19_Weights.IMAGENET1K_V1).features.eval()
        for p in vgg.parameters():
            p.requires_grad_(False)
        self.vgg = vgg

    def forward(self, x):
        # replicate grayscale texture to 3 channels for VGG input
        if x.shape[1] == 1:
            x = x.repeat(1, 3, 1, 1)
        feats = {}
        for name, layer in self.vgg._modules.items():
            x = layer(x)
            if name in self.LAYER_MAP:
                feats[self.LAYER_MAP[name]] = x
            if name == "15":
                break
        return feats


def gram_matrix(feat):
    b, c, h, w = feat.shape
    f = feat.view(b, c, h * w)
    g = torch.bmm(f, f.transpose(1, 2))
    return g / (c * h * w)


def total_variation(x):
    dh = (x[:, :, 1:, :] - x[:, :, :-1, :]).abs().mean()
    dw = (x[:, :, :, 1:] - x[:, :, :, :-1]).abs().mean()
    return dh + dw


def combined_loss(vgg, output, target_style, target_content,
                   lambda_s=100.0, lambda_c=0.5, lambda_tv=1e-4):
    """Eq. (8): L_total = lambda_s * L_style + lambda_c * L_content + lambda_tv * L_TV"""
    out_feats = vgg(output)
    style_feats = vgg(target_style)
    content_feats = vgg(target_content)

    l_style = 0.0
    for layer in ["relu1_1", "relu2_1", "relu3_1"]:
        l_style = l_style + F.mse_loss(gram_matrix(out_feats[layer]), gram_matrix(style_feats[layer]))

    l_content = F.mse_loss(out_feats["relu3_3"], content_feats["relu3_3"])
    l_tv = total_variation(output)

    total = lambda_s * l_style + lambda_c * l_content + lambda_tv * l_tv
    return total, {"style": l_style.item(), "content": l_content.item(), "tv": l_tv.item()}


def refine_texture(warped_texture_np, n_steps=150, lr=1e-3, device="cpu"):
    """
    Refines a single warped procedural texture (numpy array in [0,1],
    shape HxW) toward photorealistic appearance via per-texture
    optimization, as described in Section III-C. Uses cosine-decay Adam.
    """
    vgg = VGGFeatures().to(device)
    model = LightUNet().to(device)
    opt = torch.optim.Adam(model.parameters(), lr=lr)
    sched = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=n_steps)

    x = torch.tensor(warped_texture_np, dtype=torch.float32, device=device)[None, None]
    target_style = x.clone()
    target_content = x.clone()

    for step in range(n_steps):
        opt.zero_grad()
        out = model(x)
        loss, parts = combined_loss(vgg, out, target_style, target_content)
        loss.backward()
        opt.step()
        sched.step()

    with torch.no_grad():
        result = model(x).squeeze().cpu().numpy()
    return result, count_parameters(model)


if __name__ == "__main__":
    m = LightUNet()
    print(f"U-Net parameter count: {count_parameters(m):,}")
